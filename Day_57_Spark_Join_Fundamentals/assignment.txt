from pyspark.sql import SparkSession

spark = SparkSession.builder \
    .appName("Class7A_Join_Homework") \
    .getOrCreate()


# ------------------------------------------------------------
# EMPLOYEES
# ------------------------------------------------------------

employees_data = [
    (101, "Rahul", 10, 85000),
    (102, "Priya", 20, 92000),
    (103, "Amit", 10, 78000),
    (104, "Neha", 30, 70000),
    (105, "Arjun", 40, 88000),
    (106, "Sneha", 50, 76000),
    (107, "Vikram", 60, 81000)
]

employees = spark.createDataFrame(
    employees_data,
    ["employee_id", "employee_name", "department_id", "salary"]
)


# ------------------------------------------------------------
# DEPARTMENTS
# ------------------------------------------------------------

departments_data = [
    (10, "Engineering"),
    (20, "Data Science"),
    (30, "Human Resources"),
    (40, "Finance"),
    (50, "Marketing"),
    (70, "Legal")
]

departments = spark.createDataFrame(
    departments_data,
    ["department_id", "department_name"]
)


# ------------------------------------------------------------
# PROJECTS
# ------------------------------------------------------------

projects_data = [
    (501, "Cloud Migration", 101),
    (502, "Customer Analytics", 102),
    (503, "Payment Gateway", 103),
    (504, "HR Automation", 104),
    (505, "Financial Dashboard", 105),
    (506, "Marketing Campaign", 108)
]

projects = spark.createDataFrame(
    projects_data,
    ["project_id", "project_name", "employee_id"]
)


# ------------------------------------------------------------
# OFFICES
# ------------------------------------------------------------

offices_data = [
    (10, "Bangalore"),
    (20, "Mumbai"),
    (30, "Delhi"),
    (40, "Hyderabad"),
    (50, "Pune"),
    (80, "Chennai")
]

offices = spark.createDataFrame(
    offices_data,
    ["department_id", "office_city"]
)


# Display the DataFrames

employees.show()
departments.show()
projects.show()
offices.show()
